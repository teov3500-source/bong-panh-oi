<script>

let count = 0;

const scene = document.getElementById("scene");
const countText = document.getElementById("count");
const levelText = document.getElementById("level");
const message = document.getElementById("message");
const button = document.getElementById("heartBtn");


const messages = [

"Bống vừa thả tim cho Phương Anh! 💗",

"Một trái tim nữa nè! 💕",

"Phương Anh nhận được tim rồi! 💖",

"Bống tiếp tục thả tim! 💘",

"Tim bay tới Phương Anh! 💞",

"💗 Bống lại thả tim 💗"

];


function playSound(){

    try{

        const audio =
        new (window.AudioContext ||
        window.webkitAudioContext)();

        const oscillator =
        audio.createOscillator();

        const gain =
        audio.createGain();

        oscillator.type = "sine";

        oscillator.frequency.value = 650;

        gain.gain.setValueAtTime(
            0.08,
            audio.currentTime
        );

        gain.gain.exponentialRampToValueAtTime(
            0.001,
            audio.currentTime + 0.18
        );

        oscillator.connect(gain);

        gain.connect(audio.destination);

        oscillator.start();

        oscillator.stop(
            audio.currentTime + 0.18
        );

    }catch(e){}

}


button.onclick = function(){

    count++;

    countText.textContent = count;

    levelText.textContent =
        Math.floor(count / 5);


    message.textContent =
        messages[(count - 1) % messages.length];


    message.animate(

        [
            {
                transform:
                "translate(-50%,-50%) scale(1)"
            },

            {
                transform:
                "translate(-50%,-50%) scale(1.1)"
            },

            {
                transform:
                "translate(-50%,-50%) scale(1)"
            }
        ],

        {
            duration:250
        }

    );


    playSound();


    for(let i=0;i<5;i++){

        const heart =
        document.createElement("div");

        heart.className = "heart";

        heart.textContent =
        ["❤️","💗","💕","💖"]
        [Math.floor(Math.random()*4)];


        heart.style.left =
        (
            scene.clientWidth * 0.70
            + Math.random()*60
            - 30
        ) + "px";


        heart.style.top =
        (
            scene.clientHeight * 0.58
            + Math.random()*30
        ) + "px";


        heart.style.setProperty(
            "--x",
            (Math.random()*260-130)+"px"
        );


        heart.style.setProperty(
            "--y",
            (-180-Math.random()*220)+"px"
        );


        heart.style.setProperty(
            "--r",
            (Math.random()*80-40)+"deg"
        );


        scene.appendChild(heart);


        setTimeout(
            ()=>heart.remove(),
            1500
        );

    }

};
